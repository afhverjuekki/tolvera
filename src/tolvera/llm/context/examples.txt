# Working Examples of Expert Functions

These serve as references for the LLM when generating new behaviors.

## CRITICAL: Correct Usage of particle_idx Parameter

```python
@ti.func
def expert_neighbor_interaction(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    '''Example showing CORRECT usage of particle_idx parameter'''
    # IMPORTANT: The 5th parameter 'particle_idx' is the index of the current particle
    # Always use this when you need to skip self in loops or access current particle's state
    
    force = ti.math.vec2(0.0, 0.0)
    neighbor_radius = 100.0
    
    # Loop through all particles to find neighbors
    for j in range(tv.pn):
        # CORRECT: Use particle_idx to skip self
        if j != particle_idx:  # NOT 'if j != i:' - 'i' is undefined!
            other = tv.p.field[j]
            if other.active > 0:
                diff = other.pos - pos
                dist = diff.norm()
                
                if dist < neighbor_radius:
                    # Interact with neighbor
                    force += (pos - other.pos) / (dist + 1.0) * 50.0
    
    # Can also use particle_idx for particle-specific behavior
    # Example: particle-specific random phase
    phase = particle_idx * 0.1
    force += ti.math.vec2(ti.sin(phase), ti.cos(phase)) * 10.0
    
    return force
```

## Gravity - Simple Downward Gravity

```python
@ti.func
def expert_gravity(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    '''Apply downward gravitational force'''
    gravity_strength = 500.0  # Magnitude of gravity
    return ti.math.vec2(0.0, -gravity_strength * mass)  # Negative Y for downward force
```

## Center Attraction - Particles Attracted to Screen Center

```python
@ti.func
def expert_center_attraction(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    '''Attract particles toward center of screen'''
    center = ti.math.vec2(tv.x / 2.0, tv.y / 2.0)
    to_center = center - pos
    dist = to_center.norm()
    
    force = ti.math.vec2(0.0, 0.0)
    if dist > 1.0:  # Avoid singularity at center
        direction = to_center / dist
        strength = 200.0 * ti.min(dist / 100.0, 2.0)  # Capped strength
        force = direction * strength
    
    return force
```

## Random Drift - Particles Drift Randomly

```python
@ti.func
def expert_random_drift(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    '''Apply random drift force'''
    # Use particle index for consistent randomness per particle
    angle = (ti.random() + particle_idx * 0.1) * 2 * 3.14159
    strength = 150.0
    return ti.math.vec2(ti.cos(angle), ti.sin(angle)) * strength
```

## Species Chase - Species 0 Chases Species 1 (Interaction Expert)

```python
@ti.func
def expert_chase(p1: ti.template(), p2: ti.template()) -> ti.math.vec2:
    '''Species 0 chases species 1 - for interaction kernels'''
    force = ti.math.vec2(0.0, 0.0)
    
    if p1.species == 0 and p2.species == 1 and p2.active > 0.0:
        to_target = p2.pos - p1.pos
        dist = to_target.norm()
        
        if dist > 5.0 and dist < 200.0:  # Chase within range
            direction = to_target / dist
            # Stronger force when closer (more exciting chase)
            strength = 400.0 * (1.0 - dist / 200.0)
            force = direction * strength
    
    return force
```

## Mutual Repulsion - Particles Repel Each Other

```python
@ti.func
def expert_repel(p1: ti.template(), p2: ti.template()) -> ti.math.vec2:
    '''Mutual repulsion between particles'''
    force = ti.math.vec2(0.0, 0.0)
    
    diff = p1.pos - p2.pos
    dist = diff.norm()
    
    # Only repel within radius
    if dist > 0.1 and dist < 50.0 and p2.active > 0.0:
        direction = diff / dist
        # Stronger force when closer (1/r^2 law)
        strength = 1000.0 / (dist * dist + 1.0)
        force = direction * strength
    
    return force
```

## Flocking Alignment - Velocity Matching (Single-Particle Expert)

```python
@ti.func
def expert_alignment(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    '''Align with average velocity of neighbors'''
    avg_vel = ti.math.vec2(0.0, 0.0)
    count = 0
    perception_radius = 100.0
    
    # Find all neighbors within perception radius
    for j in range(tv.pn):
        if j != particle_idx:
            other = tv.p.field[j]
            if other.species == species and other.active > 0:
                dist = (other.pos - pos).norm()
                
                if dist < perception_radius:
                    avg_vel += other.vel
                    count += 1
    
    force = ti.math.vec2(0.0, 0.0)
    if count > 0:
        avg_vel = avg_vel / float(count)
        # Steer towards average velocity
        desired = avg_vel * 100.0  # Scale to max speed
        force = desired - vel  # Steering = desired - current
        # Limit steering force
        force_mag = force.norm()
        if force_mag > 50.0:
            force = (force / force_mag) * 50.0
    
    return force
```

## Boundary Wrapping - Wrap Particles at Screen Edges

```python
@ti.func
def boundary_wrap(pos: ti.math.vec2) -> ti.math.vec2:
    '''Wrap particle position at boundaries'''
    wrapped = pos
    
    # Wrap horizontally
    if wrapped[0] < 0:
        wrapped[0] = tv.x - 1
    elif wrapped[0] >= tv.x:
        wrapped[0] = 0
    
    # Wrap vertically
    if wrapped[1] < 0:
        wrapped[1] = tv.y - 1
    elif wrapped[1] >= tv.y:
        wrapped[1] = 0
    
    return wrapped
```

## Boundary Bounce - Bounce Particles off Screen Edges

```python
@ti.func
def expert_boundary_bounce(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    '''Apply forces to bounce off boundaries'''
    force = ti.math.vec2(0.0, 0.0)
    boundary_distance = 50.0
    repulsion_strength = 1000.0
    
    # Left boundary
    if pos[0] < boundary_distance:
        force[0] = repulsion_strength * (boundary_distance - pos[0]) / boundary_distance
    # Right boundary
    elif pos[0] > tv.x - boundary_distance:
        force[0] = -repulsion_strength * (pos[0] - (tv.x - boundary_distance)) / boundary_distance
    
    # Top boundary
    if pos[1] < boundary_distance:
        force[1] = repulsion_strength * (boundary_distance - pos[1]) / boundary_distance
    # Bottom boundary
    elif pos[1] > tv.y - boundary_distance:
        force[1] = -repulsion_strength * (pos[1] - (tv.y - boundary_distance)) / boundary_distance
    
    return force
```

## Temporal Energy Cycle - Energy-Based Behavior

```python
@ti.func
def expert_energy_cycle(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    '''Behavior that changes based on particle energy level'''
    # Get particle's energy from custom state
    energy = tv.s.llm_particle.field[particle_idx].energy
    
    force = ti.math.vec2(0.0, 0.0)
    
    # High energy: move fast, explore
    if energy > 0.7:
        angle = ti.random() * 2 * 3.14159
        force = ti.math.vec2(ti.cos(angle), ti.sin(angle)) * 300.0
    # Medium energy: seek center for resources
    elif energy > 0.3:
        center = ti.math.vec2(tv.x / 2.0, tv.y / 2.0)
        to_center = center - pos
        dist = to_center.norm()
        if dist > 1.0:
            force = (to_center / dist) * 200.0
    # Low energy: slow down, conserve
    else:
        force = -vel * 10.0  # Damping force
    
    return force
```
