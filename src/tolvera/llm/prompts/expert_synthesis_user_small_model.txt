CENTER ATTRACTION:
@ti.func
def expert_center_attraction(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    center = ti.math.vec2(tv.x / 2, tv.y / 2)
    to_center = center - pos
    dist = to_center.norm()
    force = ti.math.vec2(0.0, 0.0)
    if dist > 1.0:
        force = (to_center / dist) * 200.0
    return force

def expert_species_activity(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:

DAY/NIGHT BEHAVIOR (using states):
@ti.func
def expert_day_night(pos: ti.math.vec2, vel: ti.math.vec2, mass: ti.f32, species: ti.i32, particle_idx: ti.i32) -> ti.math.vec2:
    # Access states
    day_phase = tv.s.llm_global.field[0].day_night_cycle
    speed_mult = tv.s.llm_particle.field[particle_idx].speed_multiplier
    
    # CRITICAL: Initialize force BEFORE conditionals!
    force = ti.math.vec2(0.0, 0.0)
    
    # Random movement
    angle = ti.random() * 2 * 3.14159
    base_force = ti.math.vec2(ti.cos(angle), ti.sin(angle)) * 150.0
    
    # Scale based on day/night
    if day_phase > 0.5:  # Day
        force = base_force * speed_mult
    else:  # Night
        force = base_force * 0.1
    
    return force
